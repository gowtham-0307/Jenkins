pipeline {
    agent any

    stages {
        stage('Use Credentials') {
            steps {
                withCredentials([usernamePassword(
                    credentialsId: 'Cred-Test',
                    usernameVariable: 'USER',
                    passwordVariable: 'PASS'
                )]) {
                        echo "Username: $USER"
                        echo "Password: $PASS"
                        sh "echo 'Hello' | curl -u ${USER}:${PASS} http://example.com"
                }
            }
        }
    }
}
