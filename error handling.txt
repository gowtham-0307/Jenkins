pipeline{
    agent any
    stages{
        stage ("Hello"){
            steps{
                echo "Hello World!"
            }
        }
    }
    post{
        always{
            echo "Always run"
        }
        success{
            echo "The run is successful"
        }
        failure{
            echo "The run is failure"
        }
    }
}

                    sh "dat"
